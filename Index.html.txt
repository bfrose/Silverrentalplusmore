        // Image Uploader / Modal Functions

        function handleFileUpload(event) {
            const file = event.target.files[0];
            if (file) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    document.getElementById('image-url-input').value = e.target.result;
                };
                reader.readAsDataURL(file);
            }
        }

        function saveNewImage() {
            const productId = parseInt(document.getElementById('edit-product-id').value);
            const newUrl = document.getElementById('image-url-input').value.trim();

            if (!newUrl) {
                showToast("Please upload an image or enter a valid URL.");
                return;
            }

            const product = products.find(p => p.id === productId);
            if (product) {
                product.image = newUrl;
                // Check if this is the hero image being updated
                if (productId === 1) {
                    document.getElementById('hero-main-img').src = newUrl;
                }
                renderCatalog(document.querySelector('.catalog-tab.bg-charcoal')?.textContent.toLowerCase() || 'all');
                closeImageModal();
                showToast("Product image updated successfully!");
            }
        }

        // Wishlist Functions
        function addToWishlist(productId) {
            const product = products.find(p => p.id === productId);
            if (product && !wishlist.some(item => item.id === productId)) {
                wishlist.push(product);
                updateWishlistBadge();
                showToast(`Added "${product.name}" to wishlist`);
            } else {
                showToast("Item is already in your wishlist!");
            }
        }

        function addPackageToWishlist(packageName, packagePrice) {
            if (!wishlist.some(item => item.name === packageName)) {
                wishlist.push({ id: 'pkg_' + Date.now(), name: packageName, price: packagePrice, category: 'tablescape package' });
                updateWishlistBadge();
                showToast(`Added "${packageName}" package to wishlist`);
            } else {
                showToast("Package is already in your wishlist!");
            }
        }

        function updateWishlistBadge() {
            document.getElementById('wishlist-badge').textContent = wishlist.length;
        }

        function openWishlistModal() {
            const container = document.getElementById('wishlist-items-container');
            container.innerHTML = '';

            if (wishlist.length === 0) {
                container.innerHTML = '<p class="text-xs text-silver-500 py-6 text-center">Your wishlist is currently empty. Browse the catalog to add items!</p>';
                document.getElementById('wishlist-total').textContent = '$0';
            } else {
                let total = 0;
                wishlist.forEach((item, index) => {
                    total += item.price;
                    const row = document.createElement('div');
                    row.className = 'py-3 flex items-center justify-between text-sm';
                    row.innerHTML = `
                        <div>
                            <div class="font-bold text-charcoal">${item.name}</div>

                        </button>
                    `;
                    container.appendChild(row);
                });
                document.getElementById('wishlist-total').textContent = '$' + total;
            }

            document.getElementById('wishlist-modal').classList.remove('hidden');
            document.getElementById('wishlist-modal').classList.add('flex');
